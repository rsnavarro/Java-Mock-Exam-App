interface Flyable {
    void fly();
}

interface Swimmable {
    void swim();
}

class Duck implements Flyable, Swimmable {
    @Override
    public void fly() {
        System.out.print("Flying at 100 km/h");
    }
    
    @Override
    public void swim() {
        System.out.print("Swimming at 50 km/h");
    }
}

public class Main {
    public static void main(String[] args) {
        Duck duck = new Duck();
        duck.fly();
        System.out.print(", ");
        duck.swim();
    }
}
