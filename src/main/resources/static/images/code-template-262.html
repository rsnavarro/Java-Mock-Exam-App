class Animal {
    public void speak() {
        System.out.print("Animal speaks");
    }
}

class Dog extends Animal {
    @Override
    public void speak() {
        super.speak();
        System.out.print(", Dog barks");
    }
}

public class Main {
    public static void main(String[] args) {
        Dog dog = new Dog();
        dog.speak();
    }
}
