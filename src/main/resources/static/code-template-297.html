Question 297: Custom Method Return Type Validation
// Custom annotation for validating method return types
@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
public @interface ValidateReturnType {
    Class<?> expectedType();
    String message() default "Invalid return type";
}

// Usage example
public class DataService {
    
    @ValidateReturnType(expectedType = String.class)
    public Object getUserName(int userId) {
        // This method should return String but returns Object
        if (userId > 0) {
            return "John Doe";  // String - valid
        }
        return 12345;  // Integer - potentially invalid
    }
    
    @ValidateReturnType(expectedType = List.class)
    public List<User> getUsers() {
        return Arrays.asList(new User("Alice"), new User("Bob"));
    }
}

// Runtime validation using reflection
public class ValidationAspect {
    
    public static void validateMethod(Object instance, String methodName, Object result) {
        try {
            Method method = instance.getClass().getMethod(methodName);
            ValidateReturnType annotation = method.getAnnotation(ValidateReturnType.class);
            
            if (annotation != null) {
                Class<?> expectedType = annotation.expectedType();
                Class<?> actualType = result.getClass();
                
                // Key question: What happens here depends on implementation
                if (!expectedType.isAssignableFrom(actualType)) {
                    System.out.println("Warning: " + annotation.message());
                    System.out.println("Expected: " + expectedType.getSimpleName());
                    System.out.println("Actual: " + actualType.getSimpleName());
                }
            }
        } catch (Exception e) {
            System.err.println("Validation error: " + e.getMessage());
        }
    }
}

// Test execution
public static void main(String[] args) {
    DataService service = new DataService();
    Object result = service.getUserName(1);
    
    // The behavior depends on how validation is implemented
    ValidationAspect.validateMethod(service, "getUserName", result);
}
