Question 298: Custom Annotation Processor for Dependency Injection
// 1. Custom annotation for dependency injection
@Target(ElementType.FIELD)
@Retention(RetentionPolicy.SOURCE)
public @interface AutoInject {
    String value() default "";
    boolean required() default true;
}

// 2. Usage in target classes
public class UserService {
    @AutoInject("userRepository")
    private UserRepository userRepo;
    
    @AutoInject(required = false)
    private CacheService cache;
}

// 3. Annotation Processor Implementation
@SupportedAnnotationTypes("com.example.AutoInject")
@SupportedSourceVersion(SourceVersion.RELEASE_17)
public class DependencyInjectionProcessor extends AbstractProcessor {
    
    @Override
    public boolean process(Set<? extends TypeElement> annotations, RoundEnvironment roundEnv) {
        
        for (Element element : roundEnv.getElementsAnnotatedWith(AutoInject.class)) {
            if (element.getKind() == ElementKind.FIELD) {
                VariableElement field = (VariableElement) element;
                TypeElement enclosingClass = (TypeElement) field.getEnclosingElement();
                
                // Generate dependency injection code
                generateInjectorClass(enclosingClass, field);
            }
        }
        return true;
    }
    
    private void generateInjectorClass(TypeElement targetClass, VariableElement field) {
        try {
            String packageName = processingEnv.getElementUtils()
                .getPackageOf(targetClass).getQualifiedName().toString();
            String className = targetClass.getSimpleName() + "Injector";
            
            // Create new Java source file at compile time
            JavaFileObject sourceFile = processingEnv.getFiler()
                .createSourceFile(packageName + "." + className);
                
            try (PrintWriter writer = new PrintWriter(sourceFile.openWriter())) {
                AutoInject annotation = field.getAnnotation(AutoInject.class);
                
                writer.println("package " + packageName + ";");
                writer.println();
                writer.println("// Generated by DependencyInjectionProcessor");
                writer.println("public class " + className + " {");
                writer.println();
                writer.println("    public static void inject(" + targetClass.getSimpleName() + " target) {");
                writer.println("        // Inject " + field.getSimpleName());
                writer.println("        if (target." + field.getSimpleName() + " == null) {");
                
                if (annotation.required()) {
                    writer.println("            target." + field.getSimpleName() + 
                        " = DependencyRegistry.get(" + field.asType() + ".class);");
                } else {
                    writer.println("            target." + field.getSimpleName() + 
                        " = DependencyRegistry.getOptional(" + field.asType() + ".class);");
                }
                
                writer.println("        }");
                writer.println("    }");
                writer.println("}");
            }
            
        } catch (IOException e) {
            processingEnv.getMessager().printMessage(
                Diagnostic.Kind.ERROR, "Failed to generate injector: " + e.getMessage());
        }
    }
}

// 4. Generated code example (created at compile time)
public class UserServiceInjector {
    public static void inject(UserService target) {
        if (target.userRepo == null) {
            target.userRepo = DependencyRegistry.get(UserRepository.class);
        }
        if (target.cache == null) {
            target.cache = DependencyRegistry.getOptional(CacheService.class);
        }
    }
}
